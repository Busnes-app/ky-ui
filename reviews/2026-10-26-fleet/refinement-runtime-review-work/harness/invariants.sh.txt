#!/usr/bin/env bash
# F1-F7 invariants: adapter/theme files untouched base..head; F7/F1 rule lines identical; F2 head scripts unchanged.
set -u
G=/home/yoshi/git/busnes.app
while read -r name dir base head; do
  wt=$G/$dir/.claude/worktrees/fleet-visual-refinement
  changed=$(git -C $wt diff --name-only $base $head | grep -Ev '^docs/ui-visual-refinement-2026-10-26/|\.png$')
  adapters=$(echo "$changed" | grep -E '(theme\.(ts|js)$|Appearance\.tsx$|AppHeader\.tsx$|static/js/app\.js$|themes-browser|/ky-ui/|nav\.spec|danger-button)' | tr '\n' ' ')
  echo "== $name adapters/F6/F7-spec files changed: ${adapters:-none}"
  for f in web/src/styles/theme.css; do
    git -C $wt cat-file -e $head:$f 2>/dev/null || continue
    b=$(git -C $wt show $base:$f | grep -E 'btn-danger|:root\[data-theme\]:not' | sha256sum | cut -c1-12)
    h=$(git -C $wt show $head:$f | grep -E 'btn-danger|:root\[data-theme\]:not' | sha256sum | cut -c1-12)
    echo "   F1/F7 rule lines base=$b head=$h $([ $b = $h ] && echo SAME || echo CHANGED)"
  done
  for f in internal/server/static/index.html internal/web/templates/head.html internal/web/templates/base.html; do
    git -C $wt cat-file -e $head:$f 2>/dev/null || continue
    b=$(git -C $wt show $base:$f | sed -n '/<head>/,/<\/head>/p' | sha256sum | cut -c1-12); h=$(git -C $wt show $head:$f | sed -n '/<head>/,/<\/head>/p' | sha256sum | cut -c1-12)
    echo "   F2 <head> of $f base=$b head=$h $([ $b = $h ] && echo SAME || echo CHANGED)"
  done
done <<'L'
