#!/usr/bin/env bash
# For each product: vendored ky-ui dir unchanged base..head, and per-file sha256 vs canonical ky-ui.
set -u
G=/home/yoshi/git/busnes.app
KYUI=$G/ky-ui/.claude/worktrees/fleet-appearance
echo "ky-ui fleet-appearance HEAD $(git -C $KYUI rev-parse HEAD)"
echo "ky-ui src/tokens.css $(git -C $KYUI show HEAD:src/tokens.css | sha256sum | cut -c1-16)"
while read -r name dir base head; do
  wt=$G/$dir/.claude/worktrees/fleet-visual-refinement
  vend=$(git -C $wt ls-tree -r --name-only $head | grep -E '(^|/)ky-ui/tokens\.css$' | head -1 | xargs dirname)
  changed=$(git -C $wt diff --name-only $base $head -- "$vend" | wc -l)
  echo "== $name vendored=$vend changed_base_to_head=$changed"
  for f in tokens.css navigation.css appearance.css appearance-themes.css themes.js themes-browser.js theme.js VERSION; do
    p="$vend/$f"; h=$(git -C $wt show "$head:$p" 2>/dev/null | sha256sum | cut -c1-16)
    c=$(git -C $KYUI show "HEAD:src/$f" 2>/dev/null | sha256sum | cut -c1-16)
    [ "$f" = VERSION ] && c=$(git -C $KYUI show HEAD:package.json | python3 -c 'import json,sys;print(json.load(sys.stdin)["version"])'); [ "$f" = VERSION ] && h=$(git -C $wt show "$head:$p" 2>/dev/null)
    git -C $wt cat-file -e "$head:$p" 2>/dev/null || { echo "   $f absent"; continue; }
    echo "   $f head=$h canonical=$c $([ "$h" = "$c" ] && echo MATCH || echo DIFF)"
  done
done <<'L'
