// Drive: Settings "Open Appearance" keeps the browser-local theme; the merged phone nav row with 15
// extra shared workspaces (appended client-side to the real list) keeps every destination reachable
// and reveals the current one without page scroll; admin and member (mocked role).
import { randomBytes } from 'node:crypto';
import { spawn } from 'node:child_process';
import { mkdtempSync, writeFileSync, rmSync } from 'node:fs';
import { connect } from 'node:net';
import { join } from 'node:path';
import { chromium } from '/home/yoshi/git/busnes.app/ky-ui/.claude/worktrees/fleet-appearance/node_modules/playwright/index.mjs';
const BIN = '/tmp/claude-refine-review/bin/drive', OUT = '/tmp/claude-refine-review/evidence/drive';
const PORT = '5391', base = `http://127.0.0.1:${PORT}`;
const initial = `Init-${randomBytes(12).toString('base64url')}9!`, password = `Rev-${randomBytes(12).toString('base64url')}9!`;
const busy = await new Promise(r => { const s = connect(+PORT, '127.0.0.1', () => { s.destroy(); r(true); }); s.on('error', () => r(false)); });
if (busy) throw new Error('5391 busy; refusing');
const dir = mkdtempSync('/tmp/claude-refine-review/run-drive-');
const proc = spawn(BIN, [], { cwd: dir, stdio: ['ignore', 'ignore', 'inherit'], env: { PATH: process.env.PATH, HOME: dir, KY_APP_URL: base, KY_HOST: '127.0.0.1', KY_PORT: PORT,
  KY_DB_DRIVER: 'sqlite', KY_DATA_DIR: join(dir, 'data'), KY_BACKUP_DIR: join(dir, 'backups'), KY_ADMIN_PASSWORD: initial, KY_CAPTCHA_PROVIDER: 'none' } });
const exited = new Promise(r => proc.once('exit', r));
const res = { head: '5d6bb8a0e181704b430495d1e8b92f45bf7ae0bc', checks: [] };
const check = (name, pass, detail) => { res.checks.push({ name, pass, detail }); console.log(pass ? 'PASS' : 'FAIL', name, JSON.stringify(detail)); };
try {
  for (let i = 0; ; i++) { try { if ((await fetch(`${base}/healthz`)).status < 500) break; } catch {} if (i > 240) throw new Error('nr'); await new Promise(r => setTimeout(r, 250)); }
  const login = await fetch(`${base}/api/auth/login`, { method: 'POST', headers: { 'Content-Type': 'application/json', Origin: base }, body: JSON.stringify({ username: 'admin', password: initial }) });
  const cookies = login.headers.getSetCookie().map(c => c.split(';')[0]); const csrf = cookies.find(c => c.startsWith('ky_csrf='))?.split('=')[1];
  const changed = await fetch(`${base}/api/auth/change-password`, { method: 'POST', headers: { 'Content-Type': 'application/json', Origin: base, Cookie: cookies.join('; '), 'X-CSRF-Token': csrf }, body: JSON.stringify({ current_password: initial, new_password: password }) });
  if (!changed.ok) throw new Error(`change-password ${changed.status}`);
  const browser = await chromium.launch(); res.browser = `chromium ${browser.version()}`;
  for (const role of ['admin', 'member']) for (const [device, viewport] of [['desktop', { width: 1280, height: 900 }], ['phone', { width: 390, height: 844 }]]) {
    const ctx = await browser.newContext({ baseURL: base, viewport, colorScheme: 'light', serviceWorkers: 'block' });
    await ctx.addInitScript(() => { if (!sessionStorage.getItem('seeded')) { localStorage.setItem('ky_theme', 'sky'); sessionStorage.setItem('seeded', '1'); } });
    await ctx.route('**/api/drive/workspaces*', async (route) => {
      if (!/\/api\/drive\/workspaces(\?|$)/.test(route.request().url()) || route.request().method() !== 'GET') return route.fallback();
      const r = await route.fetch(); const j = await r.json(); const arr = Array.isArray(j) ? j : (j.workspaces ?? j.items);
      const tpl = arr.find(w => w.kind === 'personal') ?? arr[0];
      for (let i = 1; i <= 15; i++) arr.push({ ...tpl, id: `00000000-0000-4000-8000-${String(i).padStart(12, '0')}`, name: `Team space ${i} with a long name`, kind: 'shared', role: 'manager' });
      await route.fulfill({ response: r, json: j });
    });
    if (role === 'member') await ctx.route('**/api/auth/me', async (route) => { const r = await route.fetch(); const j = await r.json(); (j.user ?? j).role = 'user'; await route.fulfill({ response: r, json: j }); });
    const page = await ctx.newPage();
    await page.goto('/'); await page.getByLabel('Username').fill('admin'); await page.getByLabel('Password').fill(password);
    await page.locator('form button[type=submit]').first().click(); await page.getByRole('navigation', { name: 'Workspaces' }).waitFor();
    await page.waitForTimeout(400);
    const key = `${role} ${device}`;
    const navInfo = () => page.evaluate(() => { const navs = document.querySelector('.app-navs'); const row = getComputedStyle(navs).display === 'contents' ? document.querySelector('.app-header') : navs; const btns = [...document.querySelectorAll('.app-header nav button')];
      const cur = row.querySelector('[aria-current="page"]'); const rr = row.getBoundingClientRect(), cr = cur?.getBoundingClientRect();
      return { count: btns.length, labels: btns.map(b => b.textContent.trim()).slice(-4), scrollable: row.scrollWidth > row.clientWidth || row.scrollHeight > row.clientHeight, current: cur?.textContent.trim(),
        curInRow: !!cr && cr.left >= rr.left - 1 && cr.right <= rr.right + 1 && cr.bottom <= rr.bottom + 1 && cr.top >= rr.top - 1, scrollY: window.scrollY, overflow: document.documentElement.scrollWidth - innerWidth,
        headerH: Math.round(document.querySelector('.app-header').getBoundingClientRect().height) }; });
    // Select the last shared workspace without Playwright scrolling (DOM click), then check the reveal.
    await page.evaluate(() => { const b = [...document.querySelectorAll('nav[aria-label="Workspaces"] button')].pop(); b.click(); });
    await page.waitForTimeout(300); const n1 = await navInfo();
    check(`${key}: last of 16 shared workspaces selected is revealed in the row; page not scrolled`, n1.curInRow && n1.scrollY === 0 && n1.overflow <= 0, n1);
    // Real Tab through the header: every nav button gets focus and is inside the row's box when focused.
    await page.locator('.app-brand').click(); let seen = 0, hidden = [];
    for (let i = 0; i < 60; i++) { await page.keyboard.press('Tab'); await page.waitForTimeout(250); const f = await page.evaluate(() => { const a = document.activeElement; if (!a?.closest('.app-header nav')) return a?.closest('.app-header') ? 'header' : 'out';
      const row = (document.querySelector('.app-navs') && getComputedStyle(document.querySelector('.app-navs')).display !== 'contents') ? document.querySelector('.app-navs') : document.querySelector('.app-header');
      const rr = row.getBoundingClientRect(), r = a.getBoundingClientRect(); const out = Math.max(rr.left - r.left, r.right - rr.right, rr.top - r.top, r.bottom - rr.bottom, 0); return { t: a.textContent.trim(), inside: out <= 1, out: Math.round(out), sb: getComputedStyle(row).scrollBehavior }; });
      if (f === 'out') break; if (f === 'header') continue; seen++; if (!f.inside) hidden.push(`${f.t} (${f.out}px, ${f.sb})`); }
    check(`${key}: every nav destination Tab-reachable and scrolled into view when focused`, seen === n1.count && hidden.length === 0, { seen, count: n1.count, hidden });
    if (role === 'admin') {
      await page.getByRole('button', { name: 'Settings & DB' }).click(); await page.getByRole('heading', { name: 'Settings', level: 1 }).waitFor();
      const before = await page.evaluate(() => ({ stored: localStorage.getItem('ky_theme'), applied: document.documentElement.dataset.theme, selects: document.querySelectorAll('main select.theme-selector, .page select.theme-selector').length }));
      await page.getByRole('button', { name: 'Open Appearance' }).click(); await page.getByRole('heading', { name: 'Appearance', level: 1 }).waitFor();
      const after = await page.evaluate(() => ({ stored: localStorage.getItem('ky_theme'), applied: document.documentElement.dataset.theme, checked: document.querySelector('input[type=radio]:checked')?.value, current: document.querySelector('.app-header [aria-current="page"]')?.textContent.trim() }));
      check(`${key}: Settings has no theme select; Open Appearance lands on Appearance with the saved theme checked`, before.selects === 0 && before.stored === 'sky' && after.stored === 'sky' && after.checked === 'sky' && after.applied === 'sky', { before, after });
      await page.screenshot({ path: `${OUT}-${key.replace(' ', '-')}-appearance.png` });
    }
    await ctx.close();
  }
  await browser.close();
} finally { proc.kill('SIGTERM'); await exited; rmSync(dir, { recursive: true, force: true }); writeFileSync(`${OUT}-results.json`, JSON.stringify(res, null, 1)); }
