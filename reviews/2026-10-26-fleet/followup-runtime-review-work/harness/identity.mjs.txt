const nav = async (page, name) => { await page.getByRole('button', { name, exact: true }).click(); await page.getByRole('heading', { level: 1 }).first().waitFor(); };
async function stepUp(page, acct) {
  const f = page.locator('#admin-stepup-password'); await f.waitFor(); await f.fill(acct.password);
  await page.getByRole('dialog', { name: /Confirm/ }).getByRole('button', { name: 'Continue' }).click();
  await f.waitFor({ state: 'detached', timeout: 60000 });
}
async function signInAs(page, acct) {
  const login = async () => { await page.goto('/'); await page.getByLabel('Username', { exact: true }).fill(acct.username); await page.getByLabel('Password', { exact: true }).fill(acct.password); await page.getByRole('button', { name: /sign in/i }).first().click(); };
  await login();
  const change = page.getByRole('button', { name: 'Set new password' });
  await Promise.race([page.getByRole('navigation').first().waitFor(), change.waitFor()]);
  if (acct.replacement && await change.isVisible()) {
    await page.getByLabel('Current or temporary password').fill(acct.password);
    await page.getByLabel('New password', { exact: true }).fill(acct.replacement);
    await page.getByLabel('Repeat new password').fill(acct.replacement);
    await change.click(); await stepUp(page, acct);
    acct.password = acct.replacement; acct.replacement = null;
    await change.waitFor({ state: 'detached' });
    if (!(await page.getByRole('navigation').first().isVisible())) await login();
  }
  await page.getByRole('navigation').first().waitFor();
}
async function seed(page) {
  await nav(page, 'Users');
  for (const acct of [helpdesk, member]) {
    await page.getByRole('button', { name: 'Create User' }).click();
    const form = page.getByRole('dialog', { name: 'Create User Account' });
    await form.getByPlaceholder('e.g. bob', { exact: true }).fill(acct.username);
    await form.getByPlaceholder('e.g. Bob Builder').fill(acct.username);
    // The Create form requires an email (with it blank, the first run never reached step-up); the bootstrap admin has none.
    await form.getByPlaceholder('bob@example.com').fill(`${acct.username}@example.test`);
    await form.getByPlaceholder('Blank: the user chooses one through a link').fill(acct.password);
    await form.locator('button[type=submit]').click(); await stepUp(page, admin);
    await form.waitFor({ state: 'detached' });
    await page.locator('.users-table tbody tr', { hasText: acct.username }).first().waitFor();
  }
  await page.locator('.users-table tbody tr', { hasText: 'helpdesk' }).locator('details.row-actions > summary').click();
  await page.getByRole('button', { name: 'Delegated administration for helpdesk' }).click();
  const editor = page.getByRole('dialog', { name: 'Delegation for helpdesk' });
  await editor.getByText(/No app connections yet|App owner/).last().waitFor(); await page.waitForTimeout(500);
  await editor.getByRole('checkbox', { name: /Helpdesk:/ }).check();
  await editor.getByRole('button', { name: 'Save', exact: true }).click(); await stepUp(page, admin);
  await editor.waitFor({ state: 'detached' });
}

  account: tr.querySelector('td[data-label="Account"]')?.textContent.trim(),
  manage: tr.querySelectorAll('details.row-actions > summary').length,
  visibleCells: [...tr.querySelectorAll('td')].filter((td) => getComputedStyle(td).display !== 'none').map((td) => td.dataset.label),
  emptyCells: [...tr.querySelectorAll('td')].filter((td) => td.matches(':empty')).map((td) => td.dataset.label),
})));

  tr.querySelector('td[data-label="Account"]')?.textContent.trim(), [...tr.querySelectorAll('.row-actions-menu .row-action')].map((b) => b.textContent.trim())])));
