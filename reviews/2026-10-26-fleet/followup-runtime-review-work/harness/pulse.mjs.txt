const OUT = `${S}/evidence/pulse`;
const PORT = '6013', base = `http://127.0.0.1:${PORT}`, password = 'ReviewUpdated456!';
const YARD_PORT = 6011;
const res = { checks: [], runs: {} };
const check = (name, pass, detail) => { res.checks.push({ name, pass, detail }); console.log(pass ? 'PASS' : 'FAIL', name, JSON.stringify(detail)); };
const free = (port, host = '127.0.0.1') => new Promise(r => { const s = connect(port, host, () => { s.destroy(); r(false); }); s.on('error', () => r(true)); });

const ip = Object.values(networkInterfaces()).flat().find(i => i?.family === 'IPv4' && !i.internal && /^(10\.|192\.168\.|172\.(1[6-9]|2\d|3[01])\.)/.test(i.address))?.address;
if (!ip) throw new Error('no private IPv4 for fake KyYard');
if (!(await free(YARD_PORT, ip)) || !(await free(YARD_PORT))) throw new Error(`${YARD_PORT} busy; refusing`);
const LONG_ORG = 'Northwind Regional Cooperative Facilities and Infrastructure Operations Group (Production)';
const LONG_EP = 'edge-node-07.warehouse-east.internal.example';
const LONG_CT = 'inventory-reconciliation-worker-with-an-exceptionally-long-container-name-0001';
const LONG_IMG = 'registry.example.internal:5000/platform/inventory/reconciliation-worker@sha256:9f86d081884c7d659a2feaa0c55ad015a3bf4f1b2b0b822cd15d6c15b0f00a08';
const yard = createServer((req, res) => {
  res.setHeader('Content-Type', 'application/json');
  if (req.url === '/api/service-tokens/claim') { res.end(JSON.stringify({ token: 'rev-yard-token', organization: { id: 'rev-org', name: LONG_ORG } })); return; }
  if (req.headers.authorization !== 'Bearer rev-yard-token') { res.writeHead(401); res.end(); return; }
  if (req.url.endsWith('/endpoints?limit=200')) { res.end(JSON.stringify([{ id: 'rev-ep', name: LONG_EP, runtime: 'docker', state: 'active' }])); return; }
  if (req.url.endsWith('/inventory')) { const t = new Date().toISOString(); res.end(JSON.stringify({ endpoint_id: 'rev-ep', state: 'active', observed_at: t, received_at: t, snapshot: { containers: [
    { id: 'ct-long', name: LONG_CT, state: 'running', status: 'Up 3 days (healthy)', image: LONG_IMG, health: 'healthy' },
    { id: 'ct-sparse', name: 'cron', state: 'exited', status: 'Exited (1) 2 minutes ago', image: 'alpine', exit_code: 1 }] } })); return; }
  if (req.url.endsWith('/samples')) { res.end(JSON.stringify([{ container_id: 'ct-long', observed_at: new Date().toISOString(), memory_bytes: 734003200, memory_limit: 2147483648, restart_count: 12 }])); return; }
  if (req.url.includes('/logs?')) { res.setHeader('Content-Type', 'text/plain'); res.end(''); return; }
  if (req.url.includes('/audit?')) { res.end('[]'); return; }
  res.writeHead(404); res.end();
});
await new Promise(r => yard.listen(YARD_PORT, ip, r));
const yardURL = `http://${ip}:${YARD_PORT}`;

// Geometry of every .dr-facts block on the page.
const measure = (page) => page.evaluate(() => {
  const r = (e) => e && e.getBoundingClientRect();
  return { pageOverflow: document.documentElement.scrollWidth - innerWidth, blocks: [...document.querySelectorAll('.dr-facts')].map((b) => ({
    aria: b.getAttribute('aria-label'), right: Math.round(r(b).right),
    facts: [...b.querySelectorAll(':scope > .dr-fact')].map((f) => {
      const l = f.querySelector('.dr-fact-label'), v = f.querySelector('.dr-fact-value'), n = f.querySelector('.dr-fact-note');
      return { label: l.textContent, labelW: Math.round(r(l).width), valueL: v && Math.round(r(v).left), valueR: v && Math.round(r(v).right),
        valueBelowLabel: v ? r(v).top >= r(l).bottom - 1 : null, noteL: n && Math.round(r(n).left), overflowX: Math.round(f.scrollWidth - f.clientWidth),
        valueOverflow: v ? Math.round(v.scrollWidth - v.clientWidth) : null };
    }) })) };
});

async function run(label, bin) {
  if (!(await free(+PORT))) throw new Error(`${PORT} busy; refusing`);
  const dir = mkdtempSync(`${S}/run-pulse-${label}-`);
  const proc = spawn(bin, [], { cwd: dir, stdio: ['ignore', 'ignore', 'inherit'], env: { PATH: process.env.PATH, KYPULSE_APP_URL: base, KYPULSE_HOST: '127.0.0.1', KYPULSE_PORT: PORT,
    KYPULSE_DB_DRIVER: 'sqlite', KYPULSE_DATA_DIR: join(dir, 'data'), KYPULSE_BACKUP_DIR: join(dir, 'backups'), KYPULSE_ADMIN_PASSWORD: 'ReviewInitial123!', KYPULSE_CAPTCHA_PROVIDER: 'none', KYPULSE_KYYARD_ALLOW_HTTP: 'true', KY_LOG_LEVEL: 'error' } });
  const exited = new Promise(r => proc.once('exit', r));
  const browser = await chromium.launch(); res.browser = `chromium ${browser.version()}`;
  try {
    for (let i = 0; ; i++) { try { if ((await fetch(base + '/healthz')).ok) break; } catch {} if (i > 100) throw new Error('not ready'); await new Promise(r => setTimeout(r, 100)); }
    const setup = await browser.newContext({ baseURL: base }); const rq = setup.request;
    const api = async (method, path, data) => { const csrf = (await setup.storageState()).cookies.find(c => c.name === 'ky_csrf')?.value;
      const r = await rq.fetch(base + path, { method, data, headers: csrf ? { 'X-CSRF-Token': csrf } : {} }); if (!r.ok()) throw new Error(`${method} ${path} ${r.status()} ${await r.text()}`); return r.json().catch(() => null); };
    await rq.get('/');
    await api('POST', '/api/auth/login', { username: 'admin', password: 'ReviewInitial123!' });
    await api('POST', '/api/auth/change-password', { current_password: 'ReviewInitial123!', new_password: password });
    await api('POST', '/api/auth/login', { username: 'admin', password });
    const id = (await api('POST', '/api/targets', { name: 'Paused fixture app', url: 'https://example.invalid/health', interval_sec: 60, enabled: false })).target.id;
    const longId = (await api('POST', '/api/targets', { name: 'Long yard app', url: 'https://example.invalid/h', interval_sec: 60, enabled: false, container: `rev-ep/${LONG_CT}` })).target.id;
    const sparseId = (await api('POST', '/api/targets', { name: 'Sparse yard app', url: 'https://example.invalid/s', interval_sec: 60, enabled: false, container: 'rev-ep/cron' })).target.id;
    await api('POST', '/api/kyyard/pair', { url: yardURL, pairing_code: '123456' });
    for (let i = 0; ; i++) { const c = await api('GET', '/api/kyyard/containers'); if (c?.length === 2) break; if (i > 150) throw new Error('yard never pulled'); await new Promise(r => setTimeout(r, 200)); }
    const storage = await setup.storageState(); await setup.close();
    for (const [device, viewport] of [['desktop', { width: 1280, height: 900 }], ['phone', { width: 390, height: 844 }]]) {
      const ctx = await browser.newContext({ baseURL: base, viewport, storageState: storage, colorScheme: 'light', serviceWorkers: 'block' });
      const page = await ctx.newPage();
      await page.goto('/#/backup'); await page.locator('.dr-facts').waitFor(); await page.waitForLoadState('networkidle');
      const facts = await page.evaluate(() => [...document.querySelectorAll('.dr-facts .dr-fact')].map(f => {
        const l = f.querySelector('.dr-fact-label'), s = l.querySelector('span'), b = l.querySelector('.badge'), v = f.querySelector('.dr-fact-value');
        const r = (e) => e && e.getBoundingClientRect();
        return { label: s?.textContent, badge: b?.textContent, labelBox: [Math.round(r(l).width), Math.round(r(l).height)],
          badgeSameLineAsLabel: b && s ? Math.abs(r(b).top - r(s).top) < 6 : null, badgeRightAligned: b ? Math.round(r(l).right - r(b).right) : null,
          valueLeftOfLabelRight: v ? Math.round(r(v).left - r(l).right) : null, valueBelowLabel: v ? r(v).top >= r(l).bottom - 1 : null };
      }));
      res.runs[`${label}-${device}-backup`] = facts;
      check(`${label} ${device}: all 4 Backup badges on their label line`, facts.length === 4 && facts.every(f => f.badgeSameLineAsLabel), facts.map(f => [f.label, f.badge, f.badgeSameLineAsLabel]));
      await page.locator('.dr-facts').screenshot({ path: `${OUT}-${label}-backup-facts-${device}.png` });

      for (const [name, tid] of [['appdetail', id], ['yardfacts-long', longId], ['yardfacts-sparse', sparseId]]) {
        await page.goto(`/#/apps/${tid}`); await page.locator('.dr-header h1').waitFor();
        if (name !== 'appdetail') await page.locator('.dr-facts[aria-label="KyYard container"]').waitFor({ timeout: 20000 });
        await page.waitForLoadState('networkidle');
        const m = await measure(page); res.runs[`${label}-${device}-${name}`] = m;
        const bad = m.blocks.flatMap(b => b.facts.filter(f => f.overflowX > 0 || (f.valueR ?? 0) > b.right + 1).map(f => f.label));
        check(`${label} ${device} ${name}: no page/fact overflow`, m.pageOverflow <= 0 && bad.length === 0, { pageOverflow: m.pageOverflow, bad, labelW: m.blocks.map(b => [b.aria, ...new Set(b.facts.map(f => f.labelW))]) });
        await page.screenshot({ path: `${OUT}-${label}-${name}-${device}.png`, fullPage: true });
      }
      await page.goto('/#/settings'); await page.locator('.panel .dr-facts').waitFor({ timeout: 20000 }); await page.waitForLoadState('networkidle');
      const m = await measure(page); res.runs[`${label}-${device}-yardcard`] = m;
      const bad = m.blocks.flatMap(b => b.facts.filter(f => f.overflowX > 0 || (f.valueR ?? 0) > b.right + 1).map(f => f.label));
      check(`${label} ${device} yardcard: no page/fact overflow`, m.pageOverflow <= 0 && bad.length === 0, { pageOverflow: m.pageOverflow, bad, labelW: m.blocks.map(b => [...new Set(b.facts.map(f => f.labelW))]) });
      await page.locator('.panel', { has: page.locator('.dr-facts') }).first().screenshot({ path: `${OUT}-${label}-yardcard-${device}.png` });

      if (label === 'head') {
        await page.goto(`/#/apps/${id}`); await page.locator('.dr-header h1').waitFor();
        const layout = await page.evaluate(() => ({ header: [...document.querySelectorAll('.dr-header button')].map(b => b.textContent.trim()),
          remove: [...document.querySelectorAll('.dr-remove button')].map(b => ({ text: b.textContent.trim(), cls: b.className })) }));
        check(`head ${device}: admin header has only Edit; Delete is the filled danger in Stop watching`, layout.header.join() === 'Edit' && layout.remove.length === 1 && /btn-danger/.test(layout.remove[0].cls), layout);
        const deletes = []; page.on('request', r => { if (r.method() === 'DELETE') deletes.push(r.url()); });
        let msg = null; page.once('dialog', d => { msg = d.message(); d.dismiss(); });
        await page.locator('.dr-remove .btn-danger').click(); await page.waitForTimeout(300);
        check(`head ${device}: Delete confirms; dismiss sends no DELETE`, !!msg && deletes.length === 0, { msg, deletes });
      }
      await ctx.close();
    }
  } finally { await browser.close(); proc.kill('SIGTERM'); await exited; rmSync(dir, { recursive: true, force: true }); }
}
try {
  await run('base', `${S}/bin/pulse-base`);
  await run('head', `${S}/bin/pulse-head`);
} finally { yard.close(); writeFileSync(`${OUT}-results.json`, JSON.stringify(res, null, 1)); }
