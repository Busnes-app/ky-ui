// (c) Calendar/Messages outline style (danger text on panel).
import { execFileSync } from 'node:child_process';
import { chromium } from '/home/yoshi/git/busnes.app/ky-ui/.claude/worktrees/fleet-appearance/node_modules/playwright/index.mjs';
const show = p => execFileSync('git', ['-C', '/home/yoshi/git/busnes.app/kyPulse-server/.claude/worktrees/fleet-appearance', 'show', `a955d6de:${p}`], { encoding: 'utf8' });
const OPT = {
  a_white: '.btn-danger{color:#fff}',
  b_bg: '.btn-danger{color:var(--bg)}',
  c_outline: '.btn-danger{background:var(--panel);color:var(--danger);border:1px solid var(--danger)}',
};
const b = await chromium.launch();
for (const [k, css] of Object.entries(OPT)) {
  const page = await b.newPage();
  await page.setContent(`<style>${show('web/src/styles/theme.css')}\n${show('web/src/ky-ui/tokens.css')}\n${css}*{transition:none!important}</style><main><div class="dr-actions"><button type="button" class="btn-danger">Delete</button></div></main>`);
  await page.addScriptTag({ content: show('web/src/ky-ui/themes-browser.js') });
  const rows = await page.evaluate(() => {
    const k = window.kyThemes, root = document.documentElement, btn = document.querySelector('button');
    const cv = document.createElement('canvas'); cv.width = cv.height = 1; const cx = cv.getContext('2d', { willReadFrequently: true });
    const px = c => { cx.clearRect(0, 0, 1, 1); cx.fillStyle = '#000'; cx.fillStyle = c; cx.fillRect(0, 0, 1, 1); return [...cx.getImageData(0, 0, 1, 1).data].slice(0, 3); };
    const lum = a => a.map(x => x / 255).map(x => x <= 0.04045 ? x / 12.92 : ((x + 0.055) / 1.055) ** 2.4).reduce((s, x, i) => s + x * [0.2126, 0.7152, 0.0722][i], 0);
    return k.THEME_CATALOG.filter(t => !['patina', 'cyber', 'nord', 'paper', 'oled'].includes(t.id) && !t.id.startsWith('busnes')).map(t => {
      root.removeAttribute('style'); k.applyThemeVariables(t.id, root); root.dataset.theme = t.id;
      const cs = getComputedStyle(btn); const f = lum(px(cs.color)), g = lum(px(cs.backgroundColor));
      return [t.id, +((Math.max(f, g) + 0.05) / (Math.min(f, g) + 0.05)).toFixed(2)];
    });
  });
  const min = rows.reduce((m, r) => r[1] < m[1] ? r : m);
  console.log(k, 'min', min.join(':'), '| <4.5:', rows.filter(r => r[1] < 4.5).map(r => r.join(':')).join(' ') || 'none');
  await page.close();
}
await b.close();
