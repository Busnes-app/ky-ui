// Shipped-artifact check: KyPulse committed web/dist CSS bundle (embedded in the binary) at head vs pre-F1,
// actual <button class="btn-danger"> markup as in KyYardCard/AppDetail, plus a screenshot strip.
import { execFileSync } from 'node:child_process';
import { chromium } from '/home/yoshi/git/busnes.app/ky-ui/.claude/worktrees/fleet-appearance/node_modules/playwright/index.mjs';
const repo = '/home/yoshi/git/busnes.app/kyPulse-server/.claude/worktrees/fleet-appearance';
const show = (rev, p) => execFileSync('git', ['-C', repo, 'show', `${rev}:${p}`], { encoding: 'utf8' });
const classic = show('a955d6de', 'web/src/ky-ui/themes-browser.js');
const LIGHT = ['light-matter', 'tropics', 'white-cliffs', 'sky', 'sun', 'polished-ky'];
const b = await chromium.launch();
for (const [label, rev, css] of [['head', 'a955d6de', 'web/dist/assets/index-D8LPBoJk.css'], ['preF1', '88fbeba', 'web/dist/assets/index-CdJyouto.css']]) {
  const page = await b.newPage({ viewport: { width: 760, height: 420 } });
  const strip = LIGHT.map(id => `<section data-id="${id}" class="dr-card" style="padding:8px;margin:4px;background:var(--panel)"><span style="color:var(--ink-strong)">${id}</span> <span class="dr-danger">Status: failed</span> <div class="dr-actions"><button type="button" class="btn-danger">Unpair</button><button type="button" class="btn-danger">Delete</button></div></section>`).join('');
  await page.setContent(`<style>${show(rev, css).replace(/url\([^)]*\)/g, 'none')}</style><style>*{transition:none!important}</style><main>${strip}</main>`);
  await page.addScriptTag({ content: classic });
  // One palette at a time on the root (products apply one theme per document); screenshot each section.
  const out = [];
  for (const id of LIGHT) {
    const r = await page.evaluate(id => {
      const root = document.documentElement; root.removeAttribute('style'); window.kyThemes.applyThemeVariables(id, root); root.dataset.theme = id;
      const cv = document.createElement('canvas'); cv.width = cv.height = 1; const cx = cv.getContext('2d', { willReadFrequently: true });
      const px = c => { cx.clearRect(0, 0, 1, 1); cx.fillStyle = '#000'; cx.fillStyle = c; cx.fillRect(0, 0, 1, 1); return [...cx.getImageData(0, 0, 1, 1).data].slice(0, 3); };
      const lum = a => a.map(x => x / 255).map(x => x <= 0.04045 ? x / 12.92 : ((x + 0.055) / 1.055) ** 2.4).reduce((s, x, i) => s + x * [0.2126, 0.7152, 0.0722][i], 0);
      const btn = document.querySelector(`[data-id="${id}"] button`), cs = getComputedStyle(btn);
      const [f, g] = [lum(px(cs.color)), lum(px(cs.backgroundColor))];
      return { id, color: cs.color, bg: cs.backgroundColor, ratio: +((Math.max(f, g) + 0.05) / (Math.min(f, g) + 0.05)).toFixed(2) };
    }, id);
    await page.locator(`[data-id="${id}"]`).screenshot({ path: `/tmp/claude-fleet-review-v2/shots/pulse-${label}-${id}.png` });
    out.push(r);
  }
  console.log(label, rev, css, out.map(r => `${r.id}:${r.ratio}`).join(' '));
  await page.close();
}
await b.close();
