// Reviewer-owned F6 check for KyRecovery: committed static bytes, 390x844, tall test page,
// Overview "View replication" DOM click: is the current tab inside the row, and is page scroll unchanged?
import { createServer } from 'node:http';
import { execFileSync } from 'node:child_process';
import { chromium } from '/home/yoshi/git/busnes.app/ky-ui/.claude/worktrees/fleet-appearance/node_modules/playwright/index.mjs';
const repo = '/home/yoshi/git/busnes.app/kyrecovery-server/.claude/worktrees/fleet-appearance';
const show = (rev, p) => { try { return execFileSync('git', ['-C', repo, 'show', `${rev}:${p}`]); } catch { return null; } };
const types = { js: 'text/javascript', css: 'text/css', html: 'text/html', svg: 'image/svg+xml', ttf: 'font/ttf', png: 'image/png' };
const b = await chromium.launch();
for (const [label, rev] of [['before-F6 ef53cd2', 'ef53cd2cd162efcff0b7679f34d7ad31f4525d9d'], ['head e0614508', 'e0614508fdadc788268c33a0e5d0b6c8b74a54f3']]) {
  const server = createServer((req, res) => {
    const url = new URL(req.url, 'http://x').pathname;
    if (url === '/') { res.writeHead(200, { 'content-type': 'text/html' }); return res.end(show(rev, 'internal/server/static/index.html')); }
    if (url.startsWith('/api/')) { res.writeHead(200, { 'content-type': 'application/json' }); return res.end('{}'); }
    const body = url.startsWith('/static/') ? show(rev, 'internal/server/static' + url.slice(7)) : null;
    if (!body) { res.writeHead(404); return res.end(); }
    res.writeHead(200, { 'content-type': types[url.split('.').pop()] ?? 'application/octet-stream' }); res.end(body);
  });
  await new Promise(r => server.listen(0, '127.0.0.1', r));
  for (const target of ['tab-replication', 'tab-custodians', 'tab-backups']) {
    const page = await b.newPage({ viewport: { width: 390, height: 844 } });
    await page.goto(`http://127.0.0.1:${server.address().port}/`, { waitUntil: 'load' });
    await page.waitForTimeout(300);
    await page.evaluate(() => { document.querySelector('main, .container, body').style.minHeight = '400vh'; });
    // Same call as the Overview buttons' onclick (document.querySelector('[data-tab=...]').click()).
    await page.evaluate(() => scrollTo(0, 400));
    const before = await page.evaluate(() => scrollY);
    await page.evaluate(t => document.querySelector(`[data-tab=${t}]`).click(), target);
    await page.waitForTimeout(200);
    const r = await page.evaluate(t => { const tab = document.querySelector(`[data-tab="${t}"]`), row = tab.parentElement.getBoundingClientRect(), it = tab.getBoundingClientRect();
      return { current: tab.getAttribute('aria-current'), inRow: it.left >= row.left - 1 && it.right <= row.right + 1, y: scrollY, rowScroll: tab.parentElement.scrollLeft, sw: tab.parentElement.scrollWidth, cw: tab.parentElement.clientWidth, rowVisible: getComputedStyle(tab.parentElement).display, ox: getComputedStyle(tab.parentElement).overflowX }; }, target);
    console.log(JSON.stringify({ label, target, before, ...r, pageScrollUnchanged: r.y === before }));
    await page.close();
  }
  server.close();
}
await b.close();
